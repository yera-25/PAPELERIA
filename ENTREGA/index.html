Matriz de Permisos por Perfil (RBAC)

Administrador: Gestion total de perfiles de usuario y contrasenas SHA-256.
Supervisor: Altas/bajas de inventario, costos, precios, arqueo y Corte Z.
Usuario / Cajero: Mostrador POS, cobro, ticket y consulta de stock (sin altas ni cortes).
1

Inicio de Sesion y Control RBAC

Modal de login obligatorio al arrancar. Acceso seguro mediante credenciales individuales validadas con SHA-256 + salt; claves maestras iniciales en el archivo CLAVES.

2

Mostrador POS y Carrito

Escaneo de codigos de barras, busqueda en tiempo real, desglose de IVA 16%, promociones por mayoreo y selector de descuentos (F3).

3

Cobro Rapido y Cambio (F4)

Modal de cobro con botones de denominaciones ($50, $100, $200, $500), calculo automatico de cambio y validacion de monto suficiente.

4

Ticket Termico Oficial

Generacion de comprobante con LOGO.jpg, folio consecutivo, identificacion del cajero, desglose de IVA y formato para 58mm y 80mm.

5

Almacen e Inventarios

Control de stock minimo, alertas de desabasto. El boton Nuevo Producto (F2) esta habilitado para Supervisor y bloqueado para Cajero.

6

Control de Caja y Corte Z

Apertura de turno, entradas/salidas, arqueo Corte Z y nuevo modulo de Devoluciones y Cambios de producto (reembolsos, canjes, Kardex, vales termicos y exportacion CSV) exclusivo para Supervisor y Admin.

7

Gestion de Perfiles y Claves

Pestana exclusiva del Administrador para crear usuarios, asignar roles y restablecer contrasenas con cifrado criptografico SHA-256 + salt.

8

Reportes y Balance Diario (Scroll Optimizado)

Balance financiero diario, productos mas vendidos, margenes brutos, ticket promedio y exportacion de datos a CSV con UTF-8 BOM, con desplazamiento vertical responsivo sin desbordamiento.

9

Historial y Reimpresion

Consulta de tickets historicos y reimpresion en plano visual superior garantizado (z-index: 10000) sin quedar oculto tras el historial.

10

Documentacion y Descarga Segura de BD

Centro de manuales en PDF (Usuario e Instalacion) y descarga protegida de la base de datos relacional SQLite (papeleria_pos.db) mediante autorizacion criptografica de Administrador.

Modelo Entidad-Relacion en Tercera Forma Normal (3FN)

10 tablas relacionales disenadas para consistencia transaccional y eliminacion de redundancias en SQLite 3:

Scripts SQL DDL Oficiales de Creacion de Tablas

-- =============================================================================
-- ESQUEMA RELACIONAL SQLITE 3FN - SISTEMA POS PAPELERIA A B C (FASE 8)
-- =============================================================================

CREATE TABLE IF NOT EXISTS categorias (
    id_categoria INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre TEXT NOT NULL UNIQUE,
    descripcion TEXT,
    activo INTEGER NOT NULL DEFAULT 1
);

CREATE TABLE IF NOT EXISTS marcas (
    id_marca INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre TEXT NOT NULL UNIQUE,
    pais_origen TEXT DEFAULT 'Mexico',
    activo INTEGER NOT NULL DEFAULT 1
);

CREATE TABLE IF NOT EXISTS unidades_medida (
    id_unidad INTEGER PRIMARY KEY AUTOINCREMENT,
    clave TEXT NOT NULL UNIQUE,
    nombre TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS proveedores (
    id_proveedor INTEGER PRIMARY KEY AUTOINCREMENT,
    rfc TEXT NOT NULL UNIQUE,
    razon_social TEXT NOT NULL,
    telefono TEXT,
    email TEXT,
    activo INTEGER NOT NULL DEFAULT 1
);

CREATE TABLE IF NOT EXISTS productos (
    id_producto INTEGER PRIMARY KEY AUTOINCREMENT,
    codigo_barras TEXT NOT NULL UNIQUE,
    sku TEXT NOT NULL UNIQUE,
    nombre TEXT NOT NULL,
    id_categoria INTEGER NOT NULL,
    id_marca INTEGER NOT NULL,
    id_unidad INTEGER NOT NULL,
    precio_compra REAL NOT NULL CHECK(precio_compra >= 0),
    precio_venta_menudeo REAL NOT NULL CHECK(precio_venta_menudeo > 0),
    precio_venta_mayoreo REAL NOT NULL CHECK(precio_venta_mayoreo > 0),
    stock_actual INTEGER NOT NULL DEFAULT 0,
    stock_minimo INTEGER NOT NULL DEFAULT 5,
    es_servicio INTEGER NOT NULL DEFAULT 0,
    permite_desglose INTEGER NOT NULL DEFAULT 0,
    activo INTEGER NOT NULL DEFAULT 1,
    FOREIGN KEY (id_categoria) REFERENCES categorias(id_categoria) ON UPDATE CASCADE ON DELETE RESTRICT,
    FOREIGN KEY (id_marca) REFERENCES marcas(id_marca) ON UPDATE CASCADE ON DELETE RESTRICT,
    FOREIGN KEY (id_unidad) REFERENCES unidades_medida(id_unidad) ON UPDATE CASCADE ON DELETE RESTRICT
);

CREATE INDEX IF NOT EXISTS idx_prod_codigo ON productos(codigo_barras);
CREATE INDEX IF NOT EXISTS idx_prod_sku ON productos(sku);

CREATE TABLE IF NOT EXISTS usuarios (
    id_usuario INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT NOT NULL UNIQUE,
    password_hash TEXT NOT NULL,
    salt TEXT NOT NULL,
    nombre_completo TEXT NOT NULL,
    rol TEXT NOT NULL CHECK(rol IN ('ADMINISTRADOR', 'SUPERVISOR', 'USUARIO', 'CAJERO')),
    activo INTEGER NOT NULL DEFAULT 1,
    ultimo_login TEXT
);

CREATE TABLE IF NOT EXISTS turnos_caja (
    id_turno INTEGER PRIMARY KEY AUTOINCREMENT,
    id_usuario INTEGER NOT NULL,
    fecha_apertura TEXT NOT NULL,
    monto_inicial REAL NOT NULL DEFAULT 0,
    fecha_cierre TEXT,
    monto_final_sistema REAL DEFAULT 0,
    monto_final_declarado REAL DEFAULT 0,
    diferencia REAL DEFAULT 0,
    estado TEXT NOT NULL DEFAULT 'ABIERTO',
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
);

CREATE TABLE IF NOT EXISTS movimientos_caja (
    id_movimiento INTEGER PRIMARY KEY AUTOINCREMENT,
    id_turno INTEGER NOT NULL,
    tipo TEXT NOT NULL CHECK(tipo IN ('ENTRADA', 'SALIDA')),
    monto REAL NOT NULL,
    motivo TEXT NOT NULL,
    fecha_hora TEXT NOT NULL,
    FOREIGN KEY (id_turno) REFERENCES turnos_caja(id_turno)
);

CREATE TABLE IF NOT EXISTS ventas (
    id_venta INTEGER PRIMARY KEY AUTOINCREMENT,
    folio_ticket TEXT NOT NULL UNIQUE,
    id_turno INTEGER NOT NULL,
    id_usuario INTEGER NOT NULL,
    fecha_hora TEXT NOT NULL,
    subtotal REAL NOT NULL,
    iva REAL NOT NULL,
    descuento REAL NOT NULL DEFAULT 0,
    descuento_porcentaje REAL NOT NULL DEFAULT 0,
    total REAL NOT NULL,
    metodo_pago TEXT NOT NULL CHECK(metodo_pago IN ('EFECTIVO', 'TARJETA', 'TRANSFERENCIA')),
    monto_recibido REAL NOT NULL,
    cambio REAL NOT NULL,
    FOREIGN KEY (id_turno) REFERENCES turnos_caja(id_turno),
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
);

CREATE TABLE IF NOT EXISTS detalle_ventas (
    id_detalle INTEGER PRIMARY KEY AUTOINCREMENT,
    id_venta INTEGER NOT NULL,
    id_producto INTEGER NOT NULL,
    cantidad REAL NOT NULL,
    precio_unitario REAL NOT NULL,
    subtotal REAL NOT NULL,
    es_mayoreo INTEGER NOT NULL DEFAULT 0,
    FOREIGN KEY (id_venta) REFERENCES ventas(id_venta) ON DELETE CASCADE,
    FOREIGN KEY (id_producto) REFERENCES productos(id_producto)
);

CREATE TABLE IF NOT EXISTS devoluciones_cambios (
    id_devolucion INTEGER PRIMARY KEY AUTOINCREMENT,
    folio TEXT NOT NULL UNIQUE,
    tipo TEXT NOT NULL CHECK(tipo IN ('DEVOLUCION', 'CAMBIO')),
    id_producto_devuelto INTEGER NOT NULL,
    nombre_producto_devuelto TEXT NOT NULL,
    cantidad_devuelta INTEGER NOT NULL CHECK(cantidad_devuelta > 0),
    precio_unitario_devuelto REAL NOT NULL,
    total_devuelto REAL NOT NULL,
    id_producto_nuevo INTEGER,
    nombre_producto_nuevo TEXT,
    cantidad_nueva INTEGER,
    precio_unitario_nuevo REAL,
    total_nuevo REAL,
    diferencia REAL NOT NULL DEFAULT 0,
    motivo TEXT NOT NULL,
    autorizado_por TEXT NOT NULL,
    fecha_hora TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_producto_devuelto) REFERENCES productos(id_producto),
    FOREIGN KEY (id_producto_nuevo) REFERENCES productos(id_producto)
);
